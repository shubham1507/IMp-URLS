# Start from the frontend repository root
Set-Location client

# Build into the directory expected by server.js
npm run build -- --outDir build
if ($LASTEXITCODE -ne 0) { throw "Build failed. Share the error." }

if (-not (Test-Path build/index.html)) {
    throw "client/build/index.html is missing"
}

# Prepare and start the actual Express server
Set-Location ../server

npm install --omit=dev
if ($LASTEXITCODE -ne 0) { throw "Server dependency installation failed" }

$env:PORT = "8080"
node server.js
